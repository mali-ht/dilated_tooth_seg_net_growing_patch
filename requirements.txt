# =============================================================================================
# MIGRATED 2026-08-22: torch 2.1.0+cu121 -> torch 2.8.0+cu128, for Blackwell (sm_120) GPUs.
#
# WHY: CUDA 12.1 emits native SASS for sm_50..sm_90 only. On Blackwell hardware (RTX PRO 6000
# Blackwell and RTX 5090 are both sm_120) the old pins install cleanly and even report
# torch.cuda.is_available() == True, then fail on real kernel launches - cuBLAS/cuDNN ship
# precompiled SASS that cannot PTX-JIT forward. torch 2.8.0+cu128 carries native sm_120 SASS.
# Docs/SETUP.md called this out in advance under "GPUs newer than CUDA 12.1".
#
# These pins are coupled to setup_server.sh's CUDA_VERSION / GCC_VERSION / MIN_DRIVER /
# PY_VERSION constants and its Phase 7 verification block. Change them together or not at all.
#
# ALSO REMOVED 2026-08-22: torchaudio, torchtext, torchdata, torchvision, pandas, seaborn,
# mlxtend, nltk, jupyterlab, ipywidgets, streamlit, scikit-learn, scikit-image, opencv-python,
# Pillow and imgaug. A repo-wide import scan (all tracked .py; there are no notebooks) shows not
# one of them is imported anywhere. They were dead weight inherited from the upstream template,
# and several of them - imgaug and scikit-image especially - are what made the numpy/numba
# co-pinning painful. Re-add individually if a future module genuinely needs one.
# =============================================================================================

torch==2.8.0+cu128

# --- numeric core ---------------------------------------------------------------------------
# numpy stays on the 1.x line deliberately. The code was written against 1.24 and a scan found
# no numpy-2-removed APIs, but 1.26.4 keeps the blast radius at zero while still satisfying
# numba 0.61.2 (numpy>=1.24,<2.3) and scipy 1.15.3 (numpy>=1.23.5).
numpy==1.26.4
scipy==1.15.3

# REQUIRED, not optional: models/patch_collate.py imports numba at module level, so every
# train_patch_network*.py run dies at import with ModuleNotFoundError without it. It was missing
# from this file until 2026-08-20 even though the code has always depended on it.
# 0.61.2 is the numba contemporary with torch 2.8; it supports python 3.11 and numpy 1.26.
numba==0.61.2

# --- training -------------------------------------------------------------------------------
lightning==2.5.6
torchmetrics==1.7.4
tensorboard==2.20.0

# NOTE: the `setuptools<81` pin that lived here is GONE. It existed because lightning 2.1.0
# called pkg_resources.declare_namespace() at import time, which setuptools 82 removed.
# lightning 2.5.6 does not touch pkg_resources, so the pin (and the deprecation warning it
# printed on every single run) is no longer needed.
